     if(!autenticar(req,env))return error('No autorizado.',401);
     if(p==='/api/bot/register')return register(req,env);
     if(p==='/api/bot/claim')return claim(env);
     if(p==='/api/bot/ack')return ack(req,env);
     if(p==='/api/bot/cleanup'){await cleanup(env);return respuesta({ok:true})}
   }
   return error('Ruta desconocida.',404);
  }catch(e){console.error('MANZANITA-WEB: error de servidor (detalles de confesiones omitidos)');return error('Error temporal del servidor.',500)}
 }
